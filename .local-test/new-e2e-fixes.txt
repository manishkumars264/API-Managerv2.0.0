
test('imports cURL query params, preserves literal email, saves updates, and duplicates within the collection', async () => {
  const dataDir = await directory(); let { app, page } = await launch(dataDir);
  try {
    await importText(page, `curl '${baseUrl}/echo?email=user@example.com&repeat=one&repeat=two'`);
    const fields = page.locator('.request-editor .kv-row').filter({ has: page.getByRole('textbox', { name: 'Key', exact: true }) });
    await expect(fields).toHaveCount(3);
    await expect(fields.nth(0).getByRole('textbox', { name: 'Value', exact: true })).toHaveValue('user@example.com');
    await fields.nth(1).getByRole('textbox', { name: 'Value', exact: true }).fill('updated');
    await expect(page.getByRole('textbox', { name: 'Request URL', exact: true })).toHaveValue(`${baseUrl}/echo?email=user@example.com&repeat=updated&repeat=two`);
    await page.getByRole('button', { name: 'Save', exact: true }).click();
    const save = page.getByRole('dialog', { name: 'Save request', exact: true });
    await save.getByRole('textbox', { name: 'Request name', exact: true }).fill('Imported email request');
    await save.getByRole('button', { name: 'Create collection', exact: true }).click();
    await page.getByRole('textbox', { name: 'Collection name', exact: true }).fill('Imported APIs');
    await page.getByRole('button', { name: 'Create', exact: true }).click();
    await save.getByRole('button', { name: 'Save', exact: true }).click();
    await expect(save).toHaveCount(0);
    await expect.poll(async () => { try { return (await readWorkspace(dataDir)).collections[0]?.requests.length; } catch { return 0; } }).toBe(1);
    const savedId = (await readWorkspace(dataDir)).collections[0].requests[0].id;
    await fields.nth(0).getByRole('textbox', { name: 'Value', exact: true }).fill('changed@example.com');
    await page.getByRole('button', { name: 'Save', exact: true }).click();
    await expect(save).toHaveCount(0);
    await expect.poll(async () => (await readWorkspace(dataDir)).collections[0].requests[0].url).toContain('changed@example.com');
    expect((await readWorkspace(dataDir)).collections[0].requests[0].id).toBe(savedId);
    const tabs = await page.getByRole('tab').count();
    await page.getByRole('button', { name: 'Actions for Imported email request', exact: true }).click();
    await page.getByRole('button', { name: 'Duplicate request', exact: true }).click();
    await expect(page.locator('.request-row')).toHaveCount(2); await expect(page.getByRole('tab')).toHaveCount(tabs);
    await expect.poll(async () => (await readWorkspace(dataDir)).collections[0].requests.length).toBe(2);
    const duplicate = (await readWorkspace(dataDir)).collections[0].requests[1];
    expect(duplicate.id).not.toBe(savedId); expect(duplicate.name).toBe('Imported email request copy'); expect(duplicate.collectionId).toBe((await readWorkspace(dataDir)).collections[0].id);
    await page.getByRole('button', { name: 'Request actions' }).click();
    await expect(page.locator('.request-menu').getByRole('button', { name: 'Copy as cURL', exact: true })).toHaveCount(0);
    await expect(page.locator('.request-menu').getByRole('button', { name: 'Rename request', exact: true })).toHaveCount(0);
    await page.getByRole('button', { name: 'Duplicate tab', exact: true }).click();
    await expect(page.getByRole('tab')).toHaveCount(tabs + 1); await expect(page.locator('.request-row')).toHaveCount(2);
    await closeNormally(app); ({ app, page } = await launch(dataDir));
    await expect(page.locator('.request-row')).toHaveCount(2);
  } finally { await cleanup(app); }
});

test('shows fresh HTTP error and empty bodies and clears the old response on connection failure', async () => {
  const dataDir = await directory(); const { app, page } = await launch(dataDir);
  try {
    for (const [index, code] of [200, 404, 401, 403, 200, 204].entries()) {
      await page.getByRole('textbox', { name: 'Request URL', exact: true }).fill(`${baseUrl}/status/${code}`);
      await page.getByRole('button', { name: 'Send', exact: true }).click();
      await expect(page.locator('.response-metrics .status')).toContainText(String(code));
      await expect.poll(async () => { try { return (await readWorkspace(dataDir)).history.length; } catch { return 0; } }).toBe(index + 1);
      const body = (await readWorkspace(dataDir)).history[0].response!.body;
      if (code === 204) { expect(body).toBe(''); await expect(page.locator('.response-body .view-lines')).not.toContainText('fresh-response'); }
      else { expect(body).toContain(`fresh-response-${code}`); await expect(page.locator('.response-body .view-lines')).toContainText(`fresh-response-${code}`); }
    }
    await page.getByRole('textbox', { name: 'Request URL', exact: true }).fill(`${baseUrl}/disconnect`);
    await page.getByRole('button', { name: 'Send', exact: true }).click();
    await expect.poll(async () => (await readWorkspace(dataDir)).history.length).toBe(7);
    expect((await readWorkspace(dataDir)).history[0].response).toBeUndefined();
    await expect(page.locator('.response-metrics .status')).toHaveCount(0); await expect(page.locator('.response-body')).toHaveCount(0);
    await page.getByRole('button', { name: 'History', exact: true }).click();
    await expect(page.locator('.history-row')).toHaveCount(7);
  } finally { await cleanup(app); }
});

test('imports structured and inline collection params and duplicates a saved folder request', async () => {
  const dataDir = await directory(); const { app, page } = await launch(dataDir);
  try {
    await importText(page, JSON.stringify({ info: { name: 'Nested import', schema: 'https://schema.getpostman.com/json/collection/v2.1.0/collection.json' }, item: [{ name: 'Folder', item: [{ name: 'Query API', request: { method: 'GET', url: { raw: `${baseUrl}/echo?email=a@b.com&inline=yes&disabled=old`, query: [{ key: 'email', value: 'a@b.com', description: 'email metadata' }, { key: 'disabled', value: '{{missing}}', disabled: true }] } } }] }] }));
    await page.locator('.request-row .tree-main').click();
    await expect(page.locator('.request-editor').getByRole('textbox', { name: 'Key', exact: true })).toHaveCount(3);
    await expect(page.getByRole('textbox', { name: 'Request URL', exact: true })).toHaveValue(`${baseUrl}/echo?email=a@b.com&inline=yes&disabled=old`);
    await page.getByRole('button', { name: 'Send', exact: true }).click();
    await expect(page.locator('.response-metrics .status')).toHaveText('200 OK');
    await expect.poll(async () => { try { return JSON.parse((await readWorkspace(dataDir)).history[0].response!.body).path; } catch { return ''; } }).toBe('/echo?email=a%40b.com&inline=yes');
    const tabs = await page.getByRole('tab').count();
    await page.getByRole('button', { name: 'Actions for Query API', exact: true }).click(); await page.getByRole('button', { name: 'Duplicate request', exact: true }).click();
    await expect(page.getByRole('tab')).toHaveCount(tabs);
    await expect.poll(async () => (await readWorkspace(dataDir)).collections[0].folders[0].requests.length).toBe(2);
    const folder = (await readWorkspace(dataDir)).collections[0].folders[0];
    expect(folder.requests[1].folderId).toBe(folder.id); expect(folder.requests[1].params).toEqual(folder.requests[0].params);
  } finally { await cleanup(app); }
});
